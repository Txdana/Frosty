cmake_minimum_required(VERSION 3.20)
# Link the MSVC runtime statically so a clean Windows install does not depend on
# a separately installed Visual C++ Redistributable for frosty_core.dll.
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(FrostyCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost REQUIRED)
add_library(frosty_core SHARED src/core.cpp src/c_api.cpp src/gpu_preview.cpp)
if(WIN32)
 target_link_libraries(frosty_core PRIVATE d3d11 dxgi d3dcompiler)
endif()
target_include_directories(frosty_core PUBLIC include ${Boost_INCLUDE_DIRS})
target_compile_definitions(frosty_core PRIVATE FROSTY_CORE_EXPORTS)
if(MSVC)
 target_compile_options(frosty_core PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(frosty_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(frosty_core_tests tests/core_tests.cpp src/core.cpp)
target_include_directories(frosty_core_tests PRIVATE include ${Boost_INCLUDE_DIRS})
add_executable(frosty_ai_ops_tests tests/ai_ops_tests.cpp)
target_link_libraries(frosty_ai_ops_tests PRIVATE frosty_core)
enable_testing()
add_test(NAME frosty_core_tests COMMAND frosty_core_tests)
add_test(NAME frosty_ai_ops_tests COMMAND frosty_ai_ops_tests)
